import sys
from datetime import datetime

DEFAULT_FILE = "names.txt"


def read_names(path):
    """Читає імена з файлу (по одному в рядку), пропускаючи порожні рядки."""
    with open(path, encoding="utf-8") as f:
        return [line.strip() for line in f if line.strip()]


def main():
    path = sys.argv[1] if len(sys.argv) > 1 else DEFAULT_FILE

    try:
        names = read_names(path)
    except FileNotFoundError:
        print(f"Файл '{path}' не знайдено.")
        sys.exit(1)

    if not names:
        print(f"Файл '{path}' порожній.")
        sys.exit(1)

    for name in names:
        print(f"Привіт, {name}! Радий вас бачити.")

    now = datetime.now()
    print(f"\nПоточний час: {now.strftime('%H:%M:%S')}")
    print(f"Сьогоднішня дата: {now.strftime('%d.%m.%Y')}")


if __name__ == "__main__":
    main()
